// PROTOTYPE for #37: a project's sessions closed in the last 14 days. Not for merging.
import { ChevronRight, Columns2, Copy, GitBranch, Play, Search, Trash2 } from 'lucide-react'
import { useId, useState } from 'react'
import { AgentProviderIcon } from '@/components/icons/AgentProviderIcon'
import {
  ContextMenu,
  ContextMenuContent,
  ContextMenuItem,
  ContextMenuSeparator,
  ContextMenuTrigger
} from '@/components/ui/context-menu'
import { Tooltip, TooltipContent, TooltipTrigger } from '@/components/ui/tooltip'
import { cn } from '@/lib/class-names'
import { notify } from '@/state/notification-store'
import { AGENT_PROVIDERS } from '../../../shared/agent-api'
import type { WorkspaceFolder } from '../../../shared/workspace-api'
import {
  RECENT_FIRST_PAGE,
  RECENT_PAGE,
  RECENT_WINDOW_DAYS,
  closedAgo,
  prototypeRecent,
  usePrototypeStore,
  type RecentSession
} from './session-restore-prototype'

const rowClass =
  'type-tree-row flex h-tree-row w-full min-w-0 items-center gap-2 rounded-component px-2 text-left hover:bg-row-hover focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none'

export function useRecentSessions(
  folder: WorkspaceFolder,
  folders: WorkspaceFolder[]
): RecentSession[] {
  const started = usePrototypeStore((state) => state.started)
  const resumed = usePrototypeStore((state) => state.resumed)
  const removed = usePrototypeStore((state) => state.removed)
  return prototypeRecent(folder, folders, { started, resumed, removed })
}

function RecentRow({
  session,
  project,
  now
}: {
  session: RecentSession
  project: string
  now: number
}): React.JSX.Element {
  const resumeRecent = usePrototypeStore((state) => state.resumeRecent)
  const removeRecent = usePrototypeStore((state) => state.removeRecent)
  const name = session.title ?? session.providerSessionId.slice(0, 8)
  const identity = `${AGENT_PROVIDERS[session.provider]} ${session.provider === 'hatfield' ? '#' : ''}${session.providerSessionId}`
  const resume = () => resumeRecent(session, project)
  return (
    <li>
      <ContextMenu>
        <Tooltip>
          <ContextMenuTrigger asChild>
            <TooltipTrigger asChild>
              <button
                type="button"
                aria-label={`Resume ${name}`}
                onClick={resume}
                className={cn(rowClass, 'group text-muted-foreground hover:text-foreground')}
              >
                <AgentProviderIcon
                  provider={session.provider}
                  className="size-4 opacity-70 group-hover:opacity-100"
                />
                <span className={cn('min-w-0 flex-1 truncate', !session.title && 'type-hint')}>
                  {name}
                </span>
                {session.branch ? (
                  <span className="type-hint flex max-w-24 shrink-0 items-center gap-1">
                    <GitBranch className="size-3 shrink-0" aria-hidden />
                    <span className="truncate">{session.branch}</span>
                  </span>
                ) : null}
                <span className="type-hint shrink-0 tabular-nums group-hover:hidden">
                  {closedAgo(session.closedAt, now)}
                </span>
                <Play className="hidden size-3.5 shrink-0 group-hover:block" aria-hidden />
              </button>
            </TooltipTrigger>
          </ContextMenuTrigger>
          <TooltipContent className="max-w-72 break-words">
            {session.title ? `${session.title} · ` : ''}
            {identity} · closed {closedAgo(session.closedAt, now)}
            {closedAgo(session.closedAt, now).endsWith('y') ? '' : ' ago'}
          </TooltipContent>
        </Tooltip>
        <ContextMenuContent>
          <ContextMenuItem onSelect={resume}>
            <Play aria-hidden />
            Resume
          </ContextMenuItem>
          <ContextMenuItem onSelect={resume}>
            <Columns2 aria-hidden />
            Resume in Split
          </ContextMenuItem>
          <ContextMenuItem
            onSelect={() => void window.desktop.writeTerminalClipboard(session.providerSessionId)}
          >
            <Copy aria-hidden />
            Copy session ID
          </ContextMenuItem>
          <ContextMenuSeparator />
          <ContextMenuItem onSelect={() => removeRecent(session.id)}>
            <Trash2 aria-hidden />
            Remove from Recent
          </ContextMenuItem>
        </ContextMenuContent>
      </ContextMenu>
    </li>
  )
}

export function RecentSessions({
  folder,
  sessions
}: {
  folder: WorkspaceFolder
  sessions: RecentSession[]
}): React.JSX.Element | null {
  const [expanded, setExpanded] = useState(false)
  const listId = useId()
  const started = usePrototypeStore((state) => state.started)
  const shown = usePrototypeStore((state) => state.shownRecent[folder.path] ?? RECENT_FIRST_PAGE)
  const showMore = usePrototypeStore((state) => state.showMore)
  if (sessions.length === 0) return null
  const visible = sessions.slice(0, shown)
  const remaining = sessions.length - visible.length
  return (
    <li className="pl-5.5">
      <button
        type="button"
        aria-expanded={expanded}
        aria-controls={listId}
        onClick={() => setExpanded(!expanded)}
        className="type-hint flex h-tree-row w-full items-center gap-2 rounded-component px-1 text-muted-foreground hover:bg-row-hover focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
      >
        <ChevronRight className={cn('size-4 shrink-0', expanded && 'rotate-90')} aria-hidden />
        <span>Recent</span>
        <span className="tabular-nums">{sessions.length}</span>
      </button>
      <ul
        id={listId}
        aria-label={`${folder.name} recent sessions`}
        hidden={!expanded}
        className="pl-4"
      >
        {expanded
          ? visible.map((session) => (
              <RecentRow key={session.id} session={session} project={folder.path} now={started} />
            ))
          : null}
        {expanded && remaining > 0 ? (
          <li>
            <button
              type="button"
              onClick={() => showMore(folder.path)}
              className="type-hint flex h-tree-row items-center rounded-component px-2 text-link hover:bg-row-hover focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
            >
              Show {Math.min(RECENT_PAGE, remaining)} more
            </button>
          </li>
        ) : null}
        {expanded && remaining === 0 ? (
          <li>
            <button
              type="button"
              onClick={() => notify({ severity: 'info', title: 'Memory search arrives with #38' })}
              className="type-hint flex h-tree-row items-center gap-1.5 rounded-component px-2 text-muted-foreground hover:bg-row-hover hover:text-foreground focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
            >
              <Search className="size-3 shrink-0" aria-hidden />
              Older than {RECENT_WINDOW_DAYS} days: search memory
            </button>
          </li>
        ) : null}
      </ul>
    </li>
  )
}
